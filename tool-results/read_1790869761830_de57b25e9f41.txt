    13→/* Task 17-b). All of them build LaTeX from parameters.               */

    15→
    16→/** "+ 5" | "- 5" — joins a signed constant */
    17→const bop = (n: number): string => (n < 0 ? `- ${Math.abs(n)}` : `+ ${n}`);
    18→
    19→/** "+ 5x" | "- x" | "" (coefficient 0 omitted) — joins a signed term */
    20→const bterm = (n: number, v: string): string =>
    21→  n === 0 ? "" : `${n < 0 ? "- " : "+ "}${Math.abs(n) === 1 ? "" : Math.abs(n)}${v}`;
    22→
    23→/** LaTeX of a monic quadratic x² + px + q (zero terms omitted) */
    24→const bquad = (p: number, q: number): string =>
    25→  `x^2${p === 0 ? "" : ` ${bterm(p, "x")}`}${q === 0 ? "" : ` ${bop(q)}`}`;
    26→
    27→/** LaTeX of a linear argument "x + 3" | "x - 3" | "x" */
    28→const blin = (b: number): string => (b === 0 ? "x" : `x ${bop(b)}`);
    29→
    30→/** LaTeX of "|x - v|" handling v ≤ 0 */
    31→const babs = (v: number): string =>
    32→  v === 0 ? "|x|" : `\\left|x ${v > 0 ? "- " : "+ "}${Math.abs(v)}\\right|`;
    33→
    34→const bgcd = (a: number, b: number): number => (b === 0 ? Math.abs(a) : bgcd(b, a % b));
    35→
    36→/** LaTeX of the reduced fraction (n/d)·π, e.g. bpi(1, 6) → "\frac{\pi}{6}" */
    37→const bpi = (n: number, d: number): string => {
    38→  const g = bgcd(n, d) || 1;
    39→  const nn = n / g;
    40→  const dd = d / g;
    41→  if (dd === 1) return nn === 1 ? "\\pi" : `${nn}\\pi`;
    42→  if (nn === 1) return `\\frac{\\pi}{${dd}}`;
    43→  return `\\frac{${nn}\\pi}{${dd}}`;
    44→};
    45→
    46→/** "(a, b)" — renders an ordered pair */
    47→const bpt = (x: number, y: number): string => `(${x}, ${y})`;

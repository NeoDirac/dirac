    21→    21→/** Splits a value into mantissa (2 decimals) × 10^exp for display/checking. */
    22→    22→function sci(v: number): { man: number; exp: number; value: number } {
    23→    23→  const exp = Math.floor(Math.log10(Math.abs(v)));
    24→    24→  const man = Math.round((v / Math.pow(10, exp)) * 100) / 100;
    25→    25→  return { man, exp, value: man * Math.pow(10, exp) };
    26→    26→}

    28→    28→/** Short bilingual description of an axis direction (screen/page framing). */
    29→    29→function axisDesc(axis: "x" | "y" | "z", s: 1 | -1): { es: string; en: string } {
    30→    30→  if (axis === "x") {
    31→    31→    return s > 0
    32→    32→      ? { es: "$+x$ (a la derecha)", en: "$+x$ (to the right)" }
    33→    33→      : { es: "$-x$ (a la izquierda)", en: "$-x$ (to the left)" };
    34→    34→  }
    35→    35→  if (axis === "y") {
    36→    36→    return s > 0
    37→    37→      ? { es: "$+y$ (hacia arriba)", en: "$+y$ (upward)" }
    38→    38→      : { es: "$-y$ (hacia abajo)", en: "$-y$ (downward)" };
    39→    39→  }
    40→    40→  return s > 0
    41→    41→    ? { es: "$+z$ (hacia fuera de la página)", en: "$+z$ (out of the page)" }
    42→    42→    : { es: "$-z$ (hacia dentro de la página)", en: "$-z$ (into the page)" };
    43→    43→}

    12→/* ---------- number helpers (local conventions) ---------- */
    13→
    14→/** Round to 1 decimal — intermediate display values. */
    15→const r1 = (n: number): number => Math.round(n * 10) / 10;
    16→/** Round to 2 decimals — intermediate display values. */
    17→const r2 = (n: number): number => Math.round(n * 100) / 100;
    18→/** Remove floating-point noise (0.30000000000000004 → 0.3). */
    19→const clean = (n: number): number => Number(n.toPrecision(10));
    20→/** Round to 2 significant figures, half away from zero (answers with sigfig tolerance). */
    21→function sig2(n: number): number {
    22→  if (n === 0) return 0;
    23→  const exp = Math.floor(Math.log10(Math.abs(n)));
    24→  const f = Math.pow(10, exp - 1);
    25→  const scaled = n / f;
    26→  const r = scaled >= 0 ? Math.floor(scaled + 0.5) : Math.ceil(scaled - 0.5);
    27→  return clean(r * f);
    28→}
    29→
    30→const G_ACC = 9.8;
    31→const G_CONST = 6.67e-11;
    32→
    33→/** Split a positive number into a 2-sig-fig mantissa (1 ≤ m < 10) and exponent. */
    34→function sci2(n: number): { mant: number; exp: number } {
    35→  let exp = Math.floor(Math.log10(Math.abs(n)));
    36→  let mant = r1(n / Math.pow(10, exp));
    37→  if (mant >= 10) {
    38→    mant = 1;
    39→    exp += 1;
    40→  }
    41→  return { mant, exp };
    42→}

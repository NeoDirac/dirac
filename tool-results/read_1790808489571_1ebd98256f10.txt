    12→/* Physical constants — must match the values quoted in the statements. */
    13→const H_PLANCK = 6.626e-34; // J·s
    14→const C_LIGHT = 3e8; // m/s
    15→const EV = 1.602e-19; // J
    16→const HC = H_PLANCK * C_LIGHT; // J·m
    17→const R_H = 1.097e7; // m⁻¹ (Rydberg)
    18→
    19→/* Deterministic formatting helpers (no RNG inside). */
    20→const r1 = (v: number) => Math.round(v * 10) / 10;
    21→const r2 = (v: number) => Math.round(v * 100) / 100;
    22→
    23→/** Round to 3 significant figures. */
    24→function sig3(v: number): number {
    25→  if (v === 0) return 0;
    26→  const exp = Math.floor(Math.log10(Math.abs(v)));
    27→  return Math.round(v / Math.pow(10, exp - 2)) * Math.pow(10, exp - 2);
    28→}
    29→
    30→/** Splits a value into mantissa (2 decimals) × 10^exp for display/checking. */
    31→function sci(v: number): { man: number; exp: number; value: number } {
    32→  const exp = Math.floor(Math.log10(Math.abs(v)));
    33→  const man = Math.round((v / Math.pow(10, exp)) * 100) / 100;
    34→  return { man, exp, value: man * Math.pow(10, exp) };
    35→}
    36→
    37→/** Photon energy of a wavelength given in nm, in eV (unrounded chain). */
    38→function photonEv(lamNm: number): number {
    39→  return HC / (lamNm * 1e-9) / EV;
    40→}

    13→/* ---------- string helpers (LaTeX built from parameters) ---------- */
    14→
    15→/** "+ 5" | "- 5" — joins a signed constant */
    16→const op = (n: number): string => (n < 0 ? `- ${Math.abs(n)}` : `+ ${n}`);
    17→
    18→/** "+ 5x" | "- x" — joins a signed term (coefficient 1 omitted) */
    19→const opTerm = (n: number, v: string): string =>
    20→  `${n < 0 ? "- " : "+ "}${Math.abs(n) === 1 ? "" : Math.abs(n)}${v}`;
    21→
    22→/** "5x^2" | "-x" | "x" — leading term (coefficient 1 omitted) */
    23→const lead = (n: number, v: string): string =>
    24→  `${n < 0 ? "-" : ""}${Math.abs(n) === 1 ? "" : Math.abs(n)}${v}`;
    25→
    26→/** LaTeX polynomial from coefficients + variable names, zeros skipped */
    27→const poly = (cs: number[], vars: string[]): string => {
    28→  const parts: string[] = [];
    29→  cs.forEach((c, i) => {
    30→    if (c === 0) return;
    31→    const v = vars[i] ?? "";
    32→    if (parts.length === 0) parts.push(lead(c, v));
    33→    else parts.push(v ? opTerm(c, v) : op(c));
    34→  });
    35→  return parts.length ? parts.join(" ") : "0";
    36→};
    37→
    38→/** plain parser syntax ("5*x^2-3*x+7") for accepted expression answers */
    39→const polyAcc = (cs: number[], vars: string[]): string => {
    40→  const parts: string[] = [];
    41→  cs.forEach((c, i) => {
    42→    if (c === 0) return;
    43→    const v = vars[i] ?? "";
    44→    parts.push(v ? `${c}*${v}` : `${c}`);
    45→  });
    46→  return parts.length ? parts.join("+").replace(/\+-/g, "-") : "0";
    47→};
    48→
    49→/** "x - 3" from the root r (factor x − r) */
    50→const linFac = (r: number): string =>
    51→  r < 0 ? `x + ${-r}` : r > 0 ? `x - ${r}` : "x";
    52→
    53→/** LaTeX vertex form "(x - h)^2 + k" from the vertex (h, k) */
    54→const vertexForm = (h: number, k: number): string =>
    55→  `\\left(x ${h < 0 ? "+" : "-"} ${Math.abs(h)}\\right)^2 ${k < 0 ? "-" : "+"} ${Math.abs(k)}`;
    56→
    57→/** "x + 3"-style trailing constant: "" when zero, else " + 3" / " - 3" */
    58→const trail = (n: number): string => (n === 0 ? "" : ` ${op(n)}`);

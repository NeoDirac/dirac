    13→/* ---------- string helpers for LaTeX built from parameters ---------- */
    14→
    15→/** "+ 5" | "- 5" — joins a signed constant */
    16→const op = (n: number): string => (n < 0 ? `- ${Math.abs(n)}` : `+ ${n}`);
    17→
    18→/** "+ 5x" | "- x" | "+ 2y" — joins a signed term (coefficient 1 omitted) */
    19→const opTerm = (n: number, v: string): string =>
    20→  `${n < 0 ? "- " : "+ "}${Math.abs(n) === 1 ? "" : Math.abs(n)}${v}`;
    21→
    22→/** "5x" | "-x" | "x" — leading term (coefficient 1 omitted) */
    23→const lead = (n: number, v: string): string =>
    24→  `${n < 0 ? "-" : ""}${Math.abs(n) === 1 ? "" : Math.abs(n)}${v}`;
    25→
    26→/** "2x - 3" — linear expression in v with slope m and constant p */
    27→const linExpr = (m: number, p: number, v = "x"): string =>
    28→  `${lead(m, v)}${p === 0 ? "" : ` ${op(p)}`}`;
    29→
    30→/** "7 - 3" | "7 + 3" | "7" — renders a − b with the right sign */
    31→const minus = (a: number, b: number): string =>
    32→  b === 0 ? `${a}` : `${a} ${b > 0 ? "-" : "+"} ${Math.abs(b)}`;

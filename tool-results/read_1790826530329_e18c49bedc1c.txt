    16→const gcd = (a: number, b: number): number => (b === 0 ? Math.abs(a) : gcd(b, a % b));
    17→
    18→/** LaTeX for the reduced fraction (n/d)·π, e.g. piFrac(2, 3) → "\frac{2\pi}{3}" */
    19→const piFrac = (n: number, d: number): string => {
    20→  const g = gcd(n, d) || 1;
    21→  const nn = n / g;
    22→  const dd = d / g;
    23→  if (dd === 1) return nn === 1 ? "\\pi" : `${nn}\\pi`;
    24→  if (nn === 1) return `\\frac{\\pi}{${dd}}`;
    25→  return `\\frac{${nn}\\pi}{${dd}}`;
    26→};
    27→
    28→/** Plain-text expression for (n/d)·π accepted by the expression checker */
    29→const piExpr = (n: number, d: number): string => {
    30→  const g = gcd(n, d) || 1;
    31→  const nn = n / g;
    32→  const dd = d / g;
    33→  if (dd === 1) return nn === 1 ? "pi" : `${nn}*pi`;
    34→  return `${nn}*pi/${dd}`;
    35→};

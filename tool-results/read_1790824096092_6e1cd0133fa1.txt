    65→/** A topic needs attention once there is enough evidence: at least 3 attempts
    66→ *  and under 50% first-try accuracy. */
    67→function isWeakTopic(attempts: number, firstTryCorrect: number): boolean {
    68→  return attempts >= 3 && attempts > 0 && firstTryCorrect / attempts < 0.5;
    69→}

    71→function timeAgo(ts: number, lang: "es" | "en"): string {
    72→  try {
    73→    return formatDistanceToNow(new Date(ts), {
    74→      addSuffix: true,
    75→      locale: lang === "es" ? dateEs : dateEn,
    76→    });
    77→  } catch {
    78→    return "";
    79→  }
    80→}

   146→/** Consecutive calendar days (ending today, or yesterday if today is empty)
   147→ *  on which at least one problem was attempted. */
   148→function computeStreak(records: ProblemRecord[]): number {
   149→  if (records.length === 0) return 0;
   150→  const days = new Set(records.map((r) => new Date(r.timestamp).toDateString()));
   151→  const cursor = new Date();
   152→  if (!days.has(cursor.toDateString())) {
   153→    // the streak survives until the end of today only if yesterday was active
   154→    cursor.setDate(cursor.getDate() - 1);
   155→    if (!days.has(cursor.toDateString())) return 0;
   156→  }
   157→  let streak = 0;
   158→  while (days.has(cursor.toDateString())) {
   159→    streak += 1;
   160→    cursor.setDate(cursor.getDate() - 1);
   161→  }
   162→  return streak;
   163→}

   178→/** Human label for one finished session (subject · topic, or mixed). */
   179→function sessionLabel(s: SessionRecord, lang: "es" | "en"): string {
   180→  const subjectStr =
   181→    s.subjects.length === 1
   182→      ? s.subjects[0] === "math"
   183→        ? lang === "es"
   184→          ? "Matemáticas"
   185→          : "Mathematics"
   186→        : lang === "es"
   187→          ? "Física"
   188→          : "Physics"
   189→      : lang === "es"
   190→        ? "Mixto"
   191→        : "Mixed";
   192→  if (s.topicId && s.subjects.length === 1) {
   193→    const cur = s.subjects[0] === "math" ? mathCurriculum : physicsCurriculum;
   194→    const topic = cur.find((tp) => tp.id === s.topicId);
   195→    if (topic) return `${subjectStr} · ${topic.name[lang]}`;
   196→  }
   197→  return subjectStr;
   198→}

   252→  })();
   253→  const suggestionTopic = suggestion
   254→    ? (suggestion.subject === "math" ? mathCurriculum : physicsCurriculum).find(
   255→        (tp) => tp.id === suggestion.topicId,
   256→      )

   259→  const overview = [

   262→      label: t("progress.overview.firstTry"),

    23→    23→function sig3(v: number): number {
    24→    24→  if (v === 0) return 0;
    25→    25→  const exp = Math.floor(Math.log10(Math.abs(v)));
    26→    26→  return Math.round(v / Math.pow(10, exp - 2)) * Math.pow(10, exp - 2);
    27→    27→}

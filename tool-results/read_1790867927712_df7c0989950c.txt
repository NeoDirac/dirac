    19→const r1 = (v: number) => Math.round(v * 10) / 10;
    20→const r2 = (v: number) => Math.round(v * 100) / 100;
    21→
    22→/** Round to 3 significant figures. */
    23→function sig3(v: number): number {
    24→  if (v === 0) return 0;
    25→  const exp = Math.floor(Math.log10(Math.abs(v)));
    26→  return Math.round(v / Math.pow(10, exp - 2)) * Math.pow(10, exp - 2);
    27→}

   276→/** SVG progress ring — animated stroke + count in the center. */
   277→function GoalRing({ done, pct, today, goal }: { done: boolean; pct: number; today: number; goal: number }) {
   278→  const { t, formatNumber } = useI18n();
   279→  const R = 23;
   280→  const C = 2 * Math.PI * R;
   281→  return (
   282→    <svg
   283→      viewBox="0 0 56 56"
   284→      className="h-14 w-14 shrink-0"
   285→      role="img"
   286→      aria-label={`${t("goal.ringLabel")}: ${formatNumber(today)}/${formatNumber(goal)}`}
   287→    >
   288→      <circle cx="28" cy="28" r={R} fill="none" stroke="var(--border)" strokeWidth="6" />
   289→      <circle
   290→        cx="28"
   291→        cy="28"
   292→        r={R}
   293→        fill="none"
   294→        stroke={done ? "var(--success)" : "var(--primary)"}
   295→        strokeWidth="6"
   296→        strokeLinecap="round"
   297→        strokeDasharray={C}
   298→        strokeDashoffset={C * (1 - pct)}
   299→        transform="rotate(-90 28 28)"
   300→        style={{ transition: "stroke-dashoffset 700ms cubic-bezier(0.22, 1, 0.36, 1)" }}
   301→      />
   302→      {done ? (
   303→        <g fill="none" stroke="var(--success)" strokeWidth="3.5" strokeLinecap="round" strokeLinejoin="round">
   304→          <path d="M19 29l6.5 6.5L37 23.5" />

   307→        <text
   308→          x="28"
   309→          y="32.5"
   310→          textAnchor="middle"
   311→          fontSize="16"
   312→          fontWeight="700"
   313→          fill="var(--foreground)"
   314→          style={{ fontVariantNumeric: "tabular-nums" }}
   315→        >
   316→          {today}
   317→        </text>
   318→      )}
   319→    </svg>
   320→  );
   321→}

   323→/** Smart-resume band — the most recently practiced topic, one click away. */
   324→function ContinueCard({ stats }: { stats: OverallStats | null }) {
   325→  const { t, lang, formatNumber } = useI18n();
   326→
   327→  const last = (() => {
   328→    if (!stats) return null;
   329→    const entries = Object.values(stats.byTopic).filter((s) => s.attempts > 0 && s.lastTs > 0);
   330→    entries.sort((a, b) => b.lastTs - a.lastTs);
   331→    return entries[0] ?? null;
   332→  })();
   333→  if (!last) return null;
   334→
   335→  const curriculum = last.subject === "math" ? mathCurriculum : physicsCurriculum;
   336→  const topic = curriculum.find((tp) => tp.id === last.topicId);
   337→  if (!topic) return null;
   338→
   339→  const sessionLink = sessionHref({
   340→    mode: "topic",
   341→    subjects: [last.subject],
   342→    topicId: last.topicId,
   343→    difficulty: "any",
   344→    count: 10,
   345→    seed: 0,
   346→  });
   347→
   348→  let ago = "";
   349→  try {
   350→    ago = formatDistanceToNow(new Date(last.lastTs), {
   351→      addSuffix: true,
   352→      locale: lang === "es" ? dateEs : dateEn,
   353→    });
   354→  } catch {
   355→    /* date-fns guard */
   356→  }
   357→
   358→  return (
   359→    <section
   360→      aria-labelledby="continue-heading"
   361→      className="animate-in fade-in slide-in-from-bottom-2 duration-300 border-b bg-secondary/40"
   362→    >
   363→      <div className="mx-auto flex max-w-6xl flex-col items-start gap-4 px-4 py-6 sm:px-6 sm:flex-row sm:items-center">
   364→        <span className="inline-flex h-11 w-11 shrink-0 items-center justify-center rounded-xl bg-primary/10 text-primary">
   365→          <History className="h-5 w-5" aria-hidden="true" />
   366→        </span>
   367→        <div className="flex-1">
   368→          <h2 id="continue-heading" className="text-sm font-semibold uppercase tracking-wider text-muted-foreground">
   369→            {t("home.continue.title")}
   370→          </h2>
   371→          <p className="mt-1 text-lg font-medium leading-snug">
   372→            <a
   373→              href={href({ name: "topic", subject: last.subject, topicId: topic.id })}
   374→              className="transition-colors hover:text-primary"
   375→            >
   376→              {topic.name[lang]}
   377→            </a>
   378→            <span className="ml-2 text-sm font-normal text-muted-foreground">
   379→              {last.attempts === 1
   380→                ? t("progress.topic.oneAttempt")
   381→                : t("progress.topic.attempts", { n: formatNumber(last.attempts) })}
   382→              {ago ? ` · ${ago}` : ""}
   383→            </span>
   384→          </p>
   385→        </div>
   386→        <Button asChild className="shrink-0 gap-2 font-semibold">
   387→          <a href={sessionLink}>
   388→            {t("home.continue.cta")}
   389→            <ArrowRight className="h-4 w-4" aria-hidden="true" />
   390→          </a>
   391→        </Button>
   392→      </div>
   393→    </section>
   394→  );
   395→}

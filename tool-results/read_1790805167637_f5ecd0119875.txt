    13→/* ---------- number helpers (local conventions) ---------- */
    14→
    15→/** Round to 1 decimal — intermediate display values. */
    16→const r1 = (n: number): number => Math.round(n * 10) / 10;
    17→/** Round to 2 decimals — intermediate display values. */
    18→const r2 = (n: number): number => Math.round(n * 100) / 100;
    19→/** Remove floating-point noise (0.30000000000000004 → 0.3). */
    20→const clean = (n: number): number => Number(n.toPrecision(10));
    21→/** Round to 2 significant figures, half away from zero (answers with sigfig tolerance). */
    22→function sig2(n: number): number {
    23→  if (n === 0) return 0;
    24→  const exp = Math.floor(Math.log10(Math.abs(n)));
    25→  const f = Math.pow(10, exp - 1);
    26→  const scaled = n / f;
    27→  const r = scaled >= 0 ? Math.floor(scaled + 0.5) : Math.ceil(scaled - 0.5);
    28→  return clean(r * f);
    29→}
